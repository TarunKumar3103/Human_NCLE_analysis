#!/usr/bin/env python3
"""Reconstruct and analyze the human NCLE/degradation cohorts.

Primary analysis: NED versus ED.
Sensitivity analysis: NED versus ED + UN.

MobiDB disorder is intentionally not used as an inclusion criterion.
"""

from __future__ import annotations

import argparse
import json
from pathlib import Path

import numpy as np
import pandas as pd
import scipy.stats as stats


def parse_args() -> argparse.Namespace:
    parser = argparse.ArgumentParser()
    parser.add_argument("--mcshane", type=Path, required=True)
    parser.add_argument("--entanglement", type=Path, required=True)
    parser.add_argument("--plddt", type=Path, required=True)
    parser.add_argument("--knots", type=Path, required=True)
    parser.add_argument("--proteome-fasta", type=Path, required=True)
    parser.add_argument("--membrane-fasta", type=Path, required=True)
    parser.add_argument("--secreted-source", type=Path, required=True)
    parser.add_argument("--monomer-fasta", type=Path, action="append", default=[])
    parser.add_argument("--complex-annotation", type=Path)
    parser.add_argument("--output-dir", type=Path, required=True)
    return parser.parse_args()


def normalize_id(value: object) -> str:
    return str(value).strip()


def merge_ccbond(series: pd.Series) -> float:
    values: list[bool] = []
    for item in series.dropna():
        if isinstance(item, (bool, np.bool_)):
            values.append(bool(item))
        elif str(item).strip() in {"True", "1", "1.0"}:
            values.append(True)
        elif str(item).strip() in {"False", "0", "0.0"}:
            values.append(False)
    if any(values):
        return np.nan
    if values:
        return 1.0
    return 0.0


def parse_fasta(path: Path) -> dict[str, int]:
    lengths: dict[str, int] = {}
    current: str | None = None
    with path.open() as handle:
        for raw_line in handle:
            line = raw_line.strip()
            if not line:
                continue
            if line.startswith(">"):
                parts = line[1:].split("|")
                current = parts[1] if len(parts) >= 3 else parts[0].split()[0]
                current = normalize_id(current)
                lengths.setdefault(current, 0)
            elif current is not None:
                lengths[current] += len(line)
    return lengths


def parse_id_source(path: Path) -> set[str]:
    suffix = path.suffix.lower()
    if suffix in {".fa", ".faa", ".fasta"}:
        return set(parse_fasta(path))
    if suffix == ".json":
        payload = json.loads(path.read_text())
        values = payload.get("secreted", payload) if isinstance(payload, dict) else payload
        return {normalize_id(value) for value in values}
    values: set[str] = set()
    with path.open() as handle:
        for raw_line in handle:
            value = raw_line.strip().split("\t")[0].split(",")[0]
            if value and value.lower() not in {"uniprot", "uniprot_id", "entry"}:
                values.add(normalize_id(value))
    return values


def load_sources(args: argparse.Namespace) -> tuple[pd.DataFrame, dict[str, set[str]], dict[str, int]]:
    mcshane = pd.read_excel(args.mcshane, sheet_name="FinalTable_RPE_")
    mcshane = mcshane[["Protein IDs (Uniprot)", "Degradation profile"]].rename(
        columns={"Protein IDs (Uniprot)": "UniProt_ID", "Degradation profile": "Degradation_profile"}
    )
    mcshane["UniProt_ID"] = mcshane["UniProt_ID"].map(normalize_id)

    plddt = pd.read_csv(args.plddt)[["gene", "<pLDDT>", "std"]].rename(
        columns={"gene": "UniProt_ID", "<pLDDT>": "Average_pLDDT", "std": "pLDDT_SD"}
    )
    plddt["UniProt_ID"] = plddt["UniProt_ID"].map(normalize_id)
    plddt["Average_pLDDT"] = pd.to_numeric(plddt["Average_pLDDT"], errors="coerce")
    plddt["pLDDT_SD"] = pd.to_numeric(plddt["pLDDT_SD"], errors="coerce")
    if plddt["UniProt_ID"].duplicated().any():
        raise ValueError("Human_Avg_pLDDTs.csv contains duplicate UniProt IDs")

    entanglement = pd.read_csv(args.entanglement, sep="|", low_memory=False)[["gene", "CCBond"]]
    entanglement["gene"] = entanglement["gene"].map(normalize_id)
    entanglement = (
        entanglement.groupby("gene", as_index=False)
        .agg({"CCBond": merge_ccbond})
        .rename(columns={"gene": "UniProt_ID", "CCBond": "NCLE"})
    )

    knots = pd.read_table(args.knots, low_memory=False)
    knot_ids = set(
        knots.loc[knots["Organism"].eq("Human"), "Uniprot"]
        .astype(str)
        .str.strip()
        .str.split("-")
        .str[0]
    )
    proteome_lengths = parse_fasta(args.proteome_fasta)
    membrane_ids = set(parse_fasta(args.membrane_fasta))
    secreted_ids = parse_id_source(args.secreted_source)

    monomer_ids: set[str] = set()
    for path in args.monomer_fasta:
        monomer_ids.update(parse_fasta(path))

    master = mcshane.merge(plddt, on="UniProt_ID", how="inner", validate="one_to_one")
    master = master.loc[master["Average_pLDDT"] >= 70].copy()
    master = master.merge(entanglement, on="UniProt_ID", how="left", validate="one_to_one")
    master["Knotted"] = master["UniProt_ID"].isin(knot_ids).astype(int)
    master["FASTA_available"] = master["UniProt_ID"].isin(proteome_lengths).astype(int)
    master["Protein_length"] = master["UniProt_ID"].map(proteome_lengths)
    master["Membrane"] = master["UniProt_ID"].isin(membrane_ids).astype(int)
    master["Secreted"] = master["UniProt_ID"].isin(secreted_ids).astype(int)
    master["Known_monomer_annotation"] = np.where(
        master["UniProt_ID"].isin(monomer_ids), "Known_monomer", "Unclassified"
    )
    master["Entanglement_status"] = master["NCLE"].map(
        {1.0: "Native_entangled", 0.0: "Not_entangled"}
    ).fillna("Excluded_covalent_lasso")

    id_sets = {
        "knots": knot_ids,
        "membrane": membrane_ids,
        "secreted": secreted_ids,
        "known_monomer": monomer_ids,
    }
    return master, id_sets, proteome_lengths


def build_cohort(master: pd.DataFrame, profiles: list[str], name: str) -> tuple[pd.DataFrame, list[dict[str, object]]]:
    flow: list[dict[str, object]] = []
    data = master.loc[master["Degradation_profile"].isin(profiles)].copy()

    def record(step: str) -> None:
        counts = data["Degradation_profile"].value_counts()
        flow.append(
            {
                "Analysis": name,
                "Step": step,
                "Total": int(len(data)),
                "ED": int(counts.get("ED", 0)),
                "NED": int(counts.get("NED", 0)),
                "UN": int(counts.get("UN", 0)),
            }
        )

    record("pLDDT >=70 and selected degradation categories")
    data = data.loc[data["NCLE"].notna()].copy()
    record("Usable binary NCLE status")
    data = data.loc[data["Knotted"].eq(0)].copy()
    record("Exclude knotted proteins")
    data = data.loc[data["FASTA_available"].eq(1)].copy()
    record("Require FASTA sequence")
    data = data.loc[data["Membrane"].eq(0)].copy()
    record("Exclude membrane proteins")
    data = data.loc[data["Secreted"].eq(0)].copy()
    record("Exclude secreted proteins")

    data["NED_binary"] = data["Degradation_profile"].eq("NED").astype(int)
    data["z_Protein_length"] = (
        data["Protein_length"] - data["Protein_length"].mean()
    ) / data["Protein_length"].std(ddof=1)
    data["Analysis_group"] = (
        data["Degradation_profile"]
        + "_"
        + data["NCLE"].map({1.0: "Entangled", 0.0: "NotEntangled"})
    )
    data = data.sort_values("UniProt_ID").reset_index(drop=True)
    data.insert(0, "Index", np.arange(1, len(data) + 1))
    return data, flow


def crude_analysis(data: pd.DataFrame, analysis: str) -> tuple[dict[str, object], pd.DataFrame]:
    a = int(((data["NCLE"] == 1) & (data["NED_binary"] == 1)).sum())
    b = int(((data["NCLE"] == 1) & (data["NED_binary"] == 0)).sum())
    c = int(((data["NCLE"] == 0) & (data["NED_binary"] == 1)).sum())
    d = int(((data["NCLE"] == 0) & (data["NED_binary"] == 0)).sum())
    table = np.array([[a, b], [c, d]])
    odds_ratio, fisher_p = stats.fisher_exact(table, alternative="two-sided")
    log_se = np.sqrt(1 / a + 1 / b + 1 / c + 1 / d)
    lower, upper = np.exp(np.log(odds_ratio) + np.array([-1.96, 1.96]) * log_se)
    result = {
        "Analysis": analysis,
        "Method": "Crude Fisher exact",
        "Term": "NCLE",
        "N": int(len(data)),
        "Coefficient": float(np.log(odds_ratio)),
        "Odds_ratio": float(odds_ratio),
        "CI_lower": float(lower),
        "CI_upper": float(upper),
        "P_value": float(fisher_p),
    }
    counts = pd.DataFrame(
        [
            {"Analysis": analysis, "NCLE_status": "Entangled", "NED": a, "Non_NED": b, "Total": a + b},
            {"Analysis": analysis, "NCLE_status": "Not_entangled", "NED": c, "Non_NED": d, "Total": c + d},
        ]
    )
    counts["NED_frequency"] = counts["NED"] / counts["Total"]
    return result, counts


def logistic_analysis(data: pd.DataFrame, analysis: str, method: str, terms: list[str]) -> list[dict[str, object]]:
    clean = data[["NED_binary", *terms]].dropna().copy()
    term_names = ["Intercept", *terms]
    x = np.column_stack([np.ones(len(clean)), clean[terms].to_numpy(dtype=float)])
    y = clean["NED_binary"].to_numpy(dtype=float)
    beta = np.zeros(x.shape[1], dtype=float)
    for _ in range(100):
        eta = np.clip(x @ beta, -35, 35)
        probability = 1.0 / (1.0 + np.exp(-eta))
        weights = np.clip(probability * (1.0 - probability), 1e-12, None)
        information = x.T @ (weights[:, None] * x)
        score = x.T @ (y - probability)
        step = np.linalg.solve(information, score)
        beta_next = beta + step
        if np.max(np.abs(step)) < 1e-10:
            beta = beta_next
            break
        beta = beta_next
    else:
        raise RuntimeError(f"Logistic regression did not converge for {analysis}: {method}")

    eta = np.clip(x @ beta, -35, 35)
    probability = 1.0 / (1.0 + np.exp(-eta))
    weights = np.clip(probability * (1.0 - probability), 1e-12, None)
    covariance = np.linalg.inv(x.T @ (weights[:, None] * x))
    standard_errors = np.sqrt(np.diag(covariance))
    coefficients = dict(zip(term_names, beta))
    errors = dict(zip(term_names, standard_errors))
    rows: list[dict[str, object]] = []
    for term in terms:
        coefficient = float(coefficients[term])
        standard_error = float(errors[term])
        lower_coefficient = coefficient - 1.96 * standard_error
        upper_coefficient = coefficient + 1.96 * standard_error
        rows.append(
            {
                "Analysis": analysis,
                "Method": method,
                "Term": term,
                "N": int(len(clean)),
                "Coefficient": coefficient,
                "Odds_ratio": float(np.exp(coefficient)),
                "CI_lower": float(np.exp(lower_coefficient)),
                "CI_upper": float(np.exp(upper_coefficient)),
                "P_value": float(2 * stats.norm.sf(abs(coefficient / standard_error))),
            }
        )
    return rows


def add_complex_annotation(data: pd.DataFrame, path: Path) -> pd.DataFrame:
    annotation = pd.read_csv(path)
    expected = {"UniProt_ID", "Complex_member"}
    if not expected.issubset(annotation.columns):
        raise ValueError(f"Complex annotation must contain columns: {sorted(expected)}")
    annotation = annotation[["UniProt_ID", "Complex_member"]].copy()
    annotation["UniProt_ID"] = annotation["UniProt_ID"].map(normalize_id)
    annotation["Complex_member"] = pd.to_numeric(annotation["Complex_member"], errors="coerce")
    if annotation["UniProt_ID"].duplicated().any():
        raise ValueError("Complex annotation contains duplicate UniProt IDs")
    if not set(annotation["Complex_member"].dropna().unique()).issubset({0, 1}):
        raise ValueError("Complex_member must be 0, 1, or missing")
    return data.merge(annotation, on="UniProt_ID", how="left", validate="one_to_one")


def main() -> None:
    args = parse_args()
    args.output_dir.mkdir(parents=True, exist_ok=True)
    master, id_sets, proteome_lengths = load_sources(args)

    primary, flow_primary = build_cohort(master, ["ED", "NED"], "Primary: NED vs ED")
    sensitivity, flow_sensitivity = build_cohort(
        master, ["ED", "NED", "UN"], "Sensitivity: NED vs ED + UN"
    )

    results: list[dict[str, object]] = []
    count_tables: list[pd.DataFrame] = []
    for label, data in [
        ("Primary: NED vs ED", primary),
        ("Sensitivity: NED vs ED + UN", sensitivity),
    ]:
        crude, counts = crude_analysis(data, label)
        results.append(crude)
        count_tables.append(counts)
        results.extend(
            logistic_analysis(
                data,
                label,
                "Model 1: NED ~ NCLE + z(length)",
                ["NCLE", "z_Protein_length"],
            )
        )

    complex_status = "Not run: accurate binary complex-membership annotation was not supplied."
    if args.complex_annotation:
        complex_status = "Run using the supplied Complex_member annotation."
        updated: list[pd.DataFrame] = []
        for label, data in [
            ("Primary: NED vs ED", primary),
            ("Sensitivity: NED vs ED + UN", sensitivity),
        ]:
            annotated = add_complex_annotation(data, args.complex_annotation)
            results.extend(
                logistic_analysis(
                    annotated,
                    label,
                    "Model 2: NED ~ NCLE + complex + z(length)",
                    ["NCLE", "Complex_member", "z_Protein_length"],
                )
            )
            updated.append(annotated)
        primary, sensitivity = updated

    results_df = pd.DataFrame(results)
    counts_df = pd.concat(count_tables, ignore_index=True)
    flow_df = pd.DataFrame(flow_primary + flow_sensitivity)

    ordered_columns = [
        "Index",
        "UniProt_ID",
        "Degradation_profile",
        "NED_binary",
        "NCLE",
        "Entanglement_status",
        "Average_pLDDT",
        "pLDDT_SD",
        "Protein_length",
        "z_Protein_length",
        "Knotted",
        "FASTA_available",
        "Membrane",
        "Secreted",
        "Known_monomer_annotation",
        "Analysis_group",
    ]
    if "Complex_member" in primary.columns:
        ordered_columns.append("Complex_member")
    primary[ordered_columns].to_csv(args.output_dir / "primary_cohort.csv", index=False)
    sensitivity[ordered_columns].to_csv(args.output_dir / "sensitivity_cohort.csv", index=False)
    results_df.to_csv(args.output_dir / "association_results.csv", index=False)
    counts_df.to_csv(args.output_dir / "contingency_counts.csv", index=False)
    flow_df.to_csv(args.output_dir / "cohort_flow.csv", index=False)

    metadata = {
        "primary_n": int(len(primary)),
        "sensitivity_n": int(len(sensitivity)),
        "primary_group_counts": primary["Analysis_group"].value_counts().sort_index().to_dict(),
        "sensitivity_profile_ncle_counts": {
            f"{profile}|NCLE={int(ncle)}": int(count)
            for (profile, ncle), count in sensitivity.groupby(
                ["Degradation_profile", "NCLE"]
            ).size().items()
        },
        "secreted_source_ids": len(id_sets["secreted"]),
        "known_monomer_ids": len(id_sets["known_monomer"]),
        "complex_model_status": complex_status,
        "notes": [
            "MobiDB disorder was not used to select either statistical cohort.",
            "NCLE=1 denotes native non-covalent entanglement; NCLE=0 denotes no detected entanglement.",
            "Proteins with covalent lasso records are excluded from binary NCLE analyses.",
            "Known_monomer_annotation=Unclassified must not be interpreted as complex membership.",
        ],
    }
    (args.output_dir / "analysis_metadata.json").write_text(json.dumps(metadata, indent=2, default=str))

    print(json.dumps(metadata, indent=2, default=str))
    print("\nAssociation results:")
    print(results_df.to_string(index=False))

#!/usr/bin/env python3
"""Recompute human results from frozen protein rows, not stored ORs.

This is a new reproduction runner. statistical_analysis.py is the recovered
original analysis code. This runner never refreshes online annotations.
"""
import argparse
import hashlib
import importlib.metadata
import json
import sys
from datetime import datetime, timezone

import numpy as np
import pandas as pd
from scipy.optimize import minimize
from scipy.special import expit
from scipy.stats import fisher_exact, norm

ROOT = Path(__file__).resolve().parents[1]
LABELS = {
    "Primary": "Primary: NED vs ED",
    "Sensitivity": "Sensitivity: NED vs ED + UN",
}
M1 = "Model 1: NED ~ NCLE + z(length)"
M2 = "Model 2: NED ~ NCLE + complex + z(length)"
METRICS = ["N", "Coefficient", "Odds_ratio", "CI_lower", "CI_upper", "P_value"]


def sha256(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def validate_cohort(data, primary):
    if data["UniProt_ID"].isna().any() or data["UniProt_ID"].duplicated().any():
        raise ValueError("Missing or repeated UniProt ID")
    allowed = {"ED", "NED"} if primary else {"ED", "NED", "UN"}
    if not set(data["Degradation_profile"]).issubset(allowed):
        raise ValueError("Unexpected degradation category")
    for col in ["NCLE", "NED_binary"]:
        if data[col].isna().any() or not set(data[col]).issubset({0, 1}):
            raise ValueError(f"{col} must be complete and binary")
    expected_y = data["Degradation_profile"].eq("NED").astype(int)
    if not data["NED_binary"].eq(expected_y).all():
        raise ValueError("Outcome does not agree with degradation profile")
    length = pd.to_numeric(data["Protein_length"], errors="raise")
    if length.isna().any() or not length.gt(0).all() or length.std(ddof=1) <= 0:
        raise ValueError("Invalid protein lengths")
    if not data["Average_pLDDT"].ge(70).all():
        raise ValueError("A protein fails the inclusive pLDDT criterion")
    for col in ["Knotted", "Membrane", "Secreted"]:
        if not data[col].eq(0).all():
            raise ValueError(f"Excluded {col} protein in cohort")
    if not data["FASTA_available"].eq(1).all():
        raise ValueError("A FASTA sequence is missing")
    if not set(data["Complex_member"].dropna()).issubset({0, 1}):
        raise ValueError("Invalid complex-membership value")
    z = (length - length.mean()) / length.std(ddof=1)
    if not np.allclose(z, data["z_Protein_length"], atol=1e-12, rtol=1e-12):
        raise ValueError("Stored standardized lengths are inconsistent")
    # Standardize once within each full cohort, before Model 2 exclusions.
    data = data.copy()
    data["z_Protein_length"] = z
    return data


def independent_fit(data, label):
    """Independent MLE fit using a different optimizer and analytic Hessian."""
    terms = ["NCLE", "Complex_member", "z_Protein_length"]
    clean = data.dropna(subset=["NED_binary", *terms])
    x = np.column_stack([np.ones(len(clean)), clean[terms].to_numpy(float)])
    y = clean["NED_binary"].to_numpy(float)
    if np.linalg.matrix_rank(x) != x.shape[1]:
        raise ValueError("Model 2 design matrix is not full rank")

    def objective(beta):
        eta = x @ beta
        return np.logaddexp(0, eta).sum() - y @ eta

    def gradient(beta):
        return x.T @ (expit(x @ beta) - y)

    def hessian(beta):
        p = expit(x @ beta)
        return x.T @ ((p * (1 - p))[:, None] * x)

    fit = minimize(objective, np.zeros(x.shape[1]), jac=gradient,
                   hess=hessian, method="trust-exact",
                   options={"gtol": 1e-9, "maxiter": 500})
    if np.max(np.abs(gradient(fit.x))) >= 1e-6:
        raise RuntimeError(f"Independent fit did not converge: {fit.message}")
    se = np.sqrt(np.diag(np.linalg.inv(hessian(fit.x))))
    rows = []
    for i, term in enumerate(terms, 1):
        b = fit.x[i]
        rows.append(dict(Analysis=label, Method=M2, Term=term, N=len(clean),
                         Coefficient=b, Odds_ratio=np.exp(b),
                         CI_lower=np.exp(b - 1.96 * se[i]),
                         CI_upper=np.exp(b + 1.96 * se[i]),
                         P_value=2 * norm.sf(abs(b / se[i]))))
    return rows


def compare_results(actual, reference, description):
    keys = ["Analysis", "Method", "Term"]
    if actual.duplicated(keys).any() or reference.duplicated(keys).any():
        raise ValueError("Repeated result key")
    joined = actual.merge(reference, on=keys, how="outer",
                          suffixes=("_actual", "_reference"),
                          indicator=True, validate="one_to_one")
    rows = []
    for _, row in joined.iterrows():
        for metric in METRICS:
            a, b = row[f"{metric}_actual"], row[f"{metric}_reference"]
            match = row["_merge"] == "both" and np.isfinite(a) and np.isfinite(b)
            match = match and (a == b if metric == "N" else
                               np.isclose(a, b, atol=1e-8, rtol=1e-8))
            rows.append({**{k: row[k] for k in keys}, "Comparison": description,
                         "Metric": metric, "Actual": a, "Reference": b,
                         "Absolute_difference": abs(a - b), "Pass": bool(match)})
    return pd.DataFrame(rows)


def binary_diagnostic(data, exposed, comparison, label):
    y = data["NED_binary"].eq(1)
    a, b = int((exposed & y).sum()), int((exposed & ~y).sum())
    c, d = int((~exposed & y).sum()), int((~exposed & ~y).sum())
    odds, p = fisher_exact([[a, b], [c, d]], alternative="two-sided")
    return dict(Analysis=label, Comparison=comparison, Exposed_NED=a,
                Exposed_non_NED=b, Reference_NED=c, Reference_non_NED=d,
                Odds_ratio=float(odds), P_value=float(p))

"""Audit the frozen processed MobiDB export and its statistical-cohort coverage.

The export is processed output, not raw API JSON. Coordinate clipping is
reported explicitly; it does not establish sequence-version compatibility.
"""
import argparse
import json
from pathlib import Path

import numpy as np
import pandas as pd

ROOT = Path(__file__).resolve().parents[1]


def intervals(value):
    if pd.isna(value) or str(value).strip() in {"", "-"}:
        return []
    parsed = []
    for token in str(value).split(";"):
        start, end = map(int, token.strip().split("-"))
        if start > end:
            raise ValueError(f"Reversed interval: {token}")
        parsed.append((start, end))
    parsed.sort()
    merged = []
    for start, end in parsed:
        if merged and start <= merged[-1][1] + 1:
            merged[-1][1] = max(merged[-1][1], end)
        else:
            merged.append([start, end])
    return merged


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--master", type=Path, default=ROOT / "data/processed_MobiDB_master.csv")
    parser.add_argument("--reference", type=Path, default=ROOT / "data/reference_human_analysis.xlsx")
    parser.add_argument("--out", type=Path, default=ROOT / "results/disorder_audit")
    args = parser.parse_args()
    args.out.mkdir(parents=True, exist_ok=True)
    d = pd.read_csv(args.master)
    if d["UniProt_ID"].isna().any() or d["UniProt_ID"].duplicated().any():
        raise ValueError("Master identifiers must be present and unique")
    fraction = pd.to_numeric(d["Fraction_IDP_IDR"], errors="raise")
    length = pd.to_numeric(d["MobiDB_length"], errors="raise")
    count = pd.to_numeric(d["Disorder_residues"], errors="raise")
    available = fraction.notna()
    if not fraction.loc[available].between(0, 1).all():
        raise ValueError("Out-of-range fraction")
    if not length.loc[available].gt(0).all() or not count.loc[available].ge(0).all():
        raise ValueError("Invalid denominator or disorder count")
    if not np.allclose(fraction.loc[available], (count / length).loc[available], atol=1e-8, rtol=0):
        raise ValueError("Fraction disagrees with count divided by length")
    ok = d["Disorder_query_status"].eq("ok")
    if not ok.eq(available).all():
        raise ValueError("Status and fraction availability disagree")
    issues = []
    for _, row in d.loc[available].iterrows():
        regions = intervals(row["Disorder_regions"])
        full = sum(end - start + 1 for start, end in regions)
        size = int(row["MobiDB_length"])
        clipped = sum(max(0, min(size, end) - max(1, start) + 1) for start, end in regions)
        if clipped != int(row["Disorder_residues"]):
            raise ValueError(f"Clipped count does not reproduce stored count: {row['UniProt_ID']}")
        if full != clipped:
            issues.append(dict(UniProt_ID=row["UniProt_ID"], Full_interval_count=full,
                               Clipped_count=clipped, MobiDB_length=size,
                               Disorder_regions=row["Disorder_regions"]))

        cohort = pd.read_excel(args.reference, sheet_name=f"{name} Cohort")

        merged["Fraction_available"] = merged["Fraction_IDP_IDR"].notna()
        for group, sub in merged.groupby("Analysis_group"):

#!/usr/bin/env python3
"""Query MobiDB/UniProt and calculate the per-protein disorder fraction.

This production version preserves the disorder-track union used in Quyen's
sample script, while adding retries, explicit failure statuses, incremental
output, and resume support. Failed requests remain missing rather than being
misclassified as zero disorder.
"""

from __future__ import annotations

import argparse
import csv
import json
import os
import sys
import time
from collections import Counter, OrderedDict
from pathlib import Path
from typing import Iterable
from urllib.parse import quote

import requests
from requests.adapters import HTTPAdapter
from urllib3.util.retry import Retry


DISORDER_KEYWORDS = [
    "curated-disorder-priority",
    "homology-disorder-priority",
    "derived-missing_residues-priority",
    "prediction-disorder-priority",
    "derived-mobile-th_90",
]

FIELDNAMES = [
    "uniprot",
    "disorder_regions",
    "num_disorder_residues",
    "length",
    "fraction_idp_idr",
    "query_status",
    "annotation_tracks",
]


def build_session() -> requests.Session:
    retry = Retry(
        total=5,
        connect=5,
        read=5,
        status=5,
        backoff_factor=1.0,
        status_forcelist=(429, 500, 502, 503, 504),
        allowed_methods=frozenset({"GET"}),
        respect_retry_after_header=True,
        raise_on_status=False,
    )
    session = requests.Session()
    session.headers.update(
        {
            "User-Agent": "protein-decay-disorder-table/1.0",
            "Accept": "application/json",
        }
    )
    adapter = HTTPAdapter(max_retries=retry)
    session.mount("https://", adapter)
    session.mount("http://", adapter)
    return session


def request_json(session: requests.Session, url: str, timeout: int = 45):
    try:
        response = session.get(url, timeout=timeout)
    except requests.RequestException as exc:
        return None, f"request_failed:{type(exc).__name__}"

    if response.status_code == 404:
        return None, "not_found"
    if not response.ok:
        return None, f"http_{response.status_code}"

    try:
        return response.json(), None
    except (json.JSONDecodeError, ValueError):
        return None, "invalid_json"


def normalize_mobidb_payload(payload):
    if isinstance(payload, dict):
        return payload
    if isinstance(payload, list) and len(payload) == 1 and isinstance(payload[0], dict):
        return payload[0]
    return None


def get_mobidb_regions(session: requests.Session, uniprot: str):
    url = f"https://mobidb.org/api/download?format=json&acc={quote(uniprot)}"
    payload, error = request_json(session, url)
    if error:
        return None, [], f"mobidb_{error}"

    result = normalize_mobidb_payload(payload)
    if result is None:
        return None, [], "mobidb_unexpected_payload"

    data_sets = []
    tracks = []
    for key in DISORDER_KEYWORDS:
        block = result.get(key)
        if not isinstance(block, dict):
            continue
        regions = block.get("regions")
        if not isinstance(regions, list):
            continue
        tracks.append(key)
        data_sets.append(regions)

    if not tracks:
        return None, [], "mobidb_no_selected_annotation"

    return data_sets, tracks, None


def get_uniprot_length(session: requests.Session, uniprot: str):
    url = f"https://rest.uniprot.org/uniprotkb/{quote(uniprot)}.json"
    payload, error = request_json(session, url)
    if error:
        return None, f"uniprot_{error}"
    try:
        length = int(payload["sequence"]["length"])
    except (KeyError, TypeError, ValueError):
        return None, "uniprot_missing_length"
    if length <= 0:
        return None, "uniprot_invalid_length"
    return length, None


def merge_intervals(data_sets: Iterable[Iterable]) -> tuple[tuple[int, int], ...]:
    intervals = []
    for data_set in data_sets:
        for region in data_set:
            if not isinstance(region, (list, tuple)) or len(region) < 2:
                continue
            try:
                start = int(region[0])
                end = int(region[1])
            except (TypeError, ValueError):
                continue
            if start > end:
                start, end = end, start
            intervals.append((start, end))

    if not intervals:
        return ()

    intervals.sort(key=lambda item: (item[0], item[1]))
    merged = [list(intervals[0])]
    for start, end in intervals[1:]:
        if start <= merged[-1][1] + 1:
            merged[-1][1] = max(merged[-1][1], end)
        else:
            merged.append([start, end])
    return tuple((start, end) for start, end in merged)


def count_disordered_residues(regions, length: int) -> int:
    total = 0
    for start, end in regions:
        clipped_start = max(1, start)
        clipped_end = min(length, end)
        if clipped_end >= clipped_start:
            total += clipped_end - clipped_start + 1
    return total


def format_regions(regions) -> str:
    return ";".join(f"{start}-{end}" for start, end in regions) if regions else "-"


def query_one(session: requests.Session, uniprot: str) -> dict[str, str]:
    data_sets, tracks, mobidb_error = get_mobidb_regions(session, uniprot)
    length, length_error = get_uniprot_length(session, uniprot)

    record = {field: "" for field in FIELDNAMES}
    record["uniprot"] = uniprot
    record["annotation_tracks"] = ";".join(tracks)

    errors = [error for error in (mobidb_error, length_error) if error]
    if errors:
        if length is not None:
            record["length"] = str(length)
        record["query_status"] = "|".join(errors)
        return record

    regions = merge_intervals(data_sets)
    disorder_residues = count_disordered_residues(regions, length)
    fraction = disorder_residues / length
    if not 0.0 <= fraction <= 1.0:
        record["query_status"] = "invalid_fraction"
        return record

    record.update(
        {
            "disorder_regions": format_regions(regions),
            "num_disorder_residues": str(disorder_residues),
            "length": str(length),
            "fraction_idp_idr": f"{fraction:.8f}",
            "query_status": "ok",
        }
    )
    return record


def read_ids(path: Path) -> list[str]:
    seen = set()
    ids = []
    with path.open() as handle:
        for line in handle:
            accession = line.strip()
            if accession and accession not in seen:
                seen.add(accession)
                ids.append(accession)
    return ids


def read_existing(path: Path) -> OrderedDict[str, dict[str, str]]:
    records = OrderedDict()
    if not path.exists() or path.stat().st_size == 0:
        return records
    with path.open(newline="") as handle:
        reader = csv.DictReader(handle, delimiter="\t")
        if reader.fieldnames != FIELDNAMES:
            raise ValueError(
                f"Existing output has unexpected columns: {reader.fieldnames}. "
                "Use a new output filename."
            )
        for row in reader:
            accession = row.get("uniprot", "").strip()
            if accession:
                records[accession] = row
    return records


def compact_existing(path: Path, records, input_ids: list[str]) -> None:
    if not records:
        return
    tmp_path = path.with_suffix(path.suffix + ".tmp")
    with tmp_path.open("w", newline="") as handle:
        writer = csv.DictWriter(handle, fieldnames=FIELDNAMES, delimiter="\t")
        writer.writeheader()
        written = set()
        for accession in input_ids:
            if accession in records:
                writer.writerow(records[accession])
                written.add(accession)
        for accession, record in records.items():
            if accession not in written:
                writer.writerow(record)
    os.replace(tmp_path, path)

import time
import json
import argparse

import requests
from bioservices import UniProt


# MobiDB keywords in priority order
DISORDER_KEYWORDS = [
    'curated-disorder-priority',
    'homology-disorder-priority',
    'derived-missing_residues-priority',
    'prediction-disorder-priority',
    'derived-mobile-th_90',
]

MOBIDB_API_RATE_LIMIT_SEC = 0.5  # seconds between API calls


def get_mobidb_disordered_data(uniprot: str) -> list:
    """
    Retrieve disordered regions for a protein from MobiDB.

    Queries the MobiDB API for disorder annotations and returns a list of
    interval tuples ordered by the priority defined in DISORDER_KEYWORDS.

    Args:
        uniprot: UniProt accession ID.

    Returns:
        List of interval tuples, e.g. [([1, 56],), ([57, 68],)].
        Returns an empty list if the ID is not found or has no disorder data.
    """
    url = f'https://mobidb.org/api/download?format=json&acc={uniprot}'

    try:
        res = requests.get(url, timeout=30)
        res.raise_for_status()
    except requests.RequestException as e:
        print(f"[WARNING] MobiDB request failed for {uniprot}: {e}")
        return []

    try:
        result = res.json()
    except json.JSONDecodeError:
        print(f"[WARNING] Could not parse MobiDB JSON response for {uniprot}")
        return []

    disordered_regions = []
    for key in DISORDER_KEYWORDS:
        if key in result:
            regions = result[key]['regions']
            disordered_regions.append(tuple(regions))

    return disordered_regions


def merge_data(data_sets: list) -> tuple:
    """
    Merge and compact intervals from multiple disorder data sets.

    Combines all intervals, sorts by start position, merges overlapping
    intervals, and coalesces adjacent intervals.

    Args:
        data_sets: List of interval tuples as returned by get_mobidb_disordered_data.

    Returns:
        Tuple of merged [start, end] intervals, or an empty tuple if no
        valid intervals are present.
    """
    if not data_sets or isinstance(data_sets, str):
        return ()

    all_intervals = []
    for intervals in data_sets:
        all_intervals.extend(intervals)

    if not all_intervals:
        return ()

    # Sort by start position
    all_intervals.sort(key=lambda x: x[0])

    # Merge overlapping intervals
    merged = [list(all_intervals[0])]
    for interval in all_intervals[1:]:
        if interval[0] <= merged[-1][1]:
            merged[-1][1] = max(merged[-1][1], interval[1])
        else:
            merged.append(list(interval))

    # Coalesce adjacent intervals (end + 1 == next start)
    final = [merged[0]]
    for interval in merged[1:]:
        if final[-1][1] + 1 == interval[0]:
            final[-1][1] = interval[1]
        else:
            final.append(interval)

    return tuple(final)


def count_overlap(first_range: tuple, second_ranges: tuple) -> list:
    """
    Count overlapping residues between a query range and a set of disorder regions.

    Args:
        first_range: (start, end) tuple for the query range (e.g. full protein length).
        second_ranges: Tuple of [start, end] lists representing disorder regions.

    Returns:
        List of overlap counts per region. Returns an empty list if second_ranges is empty.
    """
    if not second_ranges:
        return []

    overlaps = []
    for region in second_ranges:
        start = max(first_range[0], region[0])
        end = min(first_range[1], region[1])
        overlaps.append(max(0, end - start + 1))

    return overlaps


def get_uniprot_length(uniprot: str, u: UniProt) -> int | None:
    """
    Retrieve the sequence length of a protein from UniProt.

    Args:
        uniprot: UniProt accession ID.
        u: Initialised bioservices UniProt instance.

    Returns:
        Sequence length as an integer, or None if the lookup fails.
    """
    res = u.search(uniprot, frmt="tsv", columns="length", limit=1, database='uniprotkb')
    lines = res.strip().split('\n') if res else []

    if len(lines) < 2:
        print(f"[WARNING] Could not find length for {uniprot} in UniProt")
        return None

    try:
        return int(lines[1])
    except ValueError:
        print(f"[WARNING] Unexpected length value for {uniprot}: {lines[1]!r}")
        return None


def get_disorder_summary(uniprot: str, u: UniProt) -> dict:
    """
    Compute the disorder summary for a single protein.

    Args:
        uniprot: UniProt accession ID.
        u: Initialised bioservices UniProt instance.

    Returns:
        Dict with keys:
            'uniprot'                : accession ID
            'length'                 : protein length (int) or None on failure
            'disorder_regions'       : tuple of merged [start, end] disorder intervals
            'num_disorder_residues'  : total number of disordered residues (int) or None
    """
    length = get_uniprot_length(uniprot, u)
    if length is None:
        return {
            'uniprot': uniprot,
            'length': None,
            'disorder_regions': (),
            'num_disorder_residues': None,
        }

    disorder_regions = merge_data(get_mobidb_disordered_data(uniprot))
    overlap_counts = count_overlap((1, length), disorder_regions)
    num_disorder_residues = sum(overlap_counts)

    return {
        'uniprot': uniprot,
        'length': length,
        'disorder_regions': disorder_regions,
        'num_disorder_residues': num_disorder_residues,
    }


def format_regions(disorder_regions: tuple) -> str:
    """Render merged disorder intervals as a compact string, e.g. '1-56;70-88'."""
    if not disorder_regions:
        return '-'
    return ';'.join(f"{start}-{end}" for start, end in disorder_regions)


def read_uniprot_list(path: str) -> list:
    """Read UniProt accessions from a file, one per line, skipping blank lines."""
    with open(path) as f:
        return [line.strip() for line in f if line.strip()]


def process_uniprots(uniprots: list, u: UniProt, output_file: str | None = None) -> None:
    """
    Fetch and report the disorder summary for each UniProt accession.

    Writes a tab-separated table (header + one row per accession) to stdout,
    and to output_file as well when provided. Columns:
        uniprot  disorder_regions  num_disorder_residues  length

    Args:
        uniprots: List of UniProt accession IDs.
        u: Initialised bioservices UniProt instance.
        output_file: Optional path to also write the table to.
    """
    header = "uniprot\tdisorder_regions\tnum_disorder_residues\tlength"

    out = open(output_file, 'w') if output_file else None
    try:
        print(header)
        if out:
            out.write(header + '\n')
            out.flush()

        for i, uniprot in enumerate(uniprots):
            summary = get_disorder_summary(uniprot, u)
            line = (
                f"{summary['uniprot']}\t"
                f"{format_regions(summary['disorder_regions'])}\t"
                f"{summary['num_disorder_residues']}\t"
                f"{summary['length']}"
            )
            print(line)
            if out:
                out.write(line + '\n')
                out.flush()

            # Rate-limit only between MobiDB API calls, not after the last one.
            if i < len(uniprots) - 1:
                time.sleep(MOBIDB_API_RATE_LIMIT_SEC)
    finally:
        if out:
            out.close()

def sha256(path):
    h = hashlib.sha256()
    with path.open('rb') as handle:
        for chunk in iter(lambda: handle.read(1024 * 1024), b''):
            h.update(chunk)
    return h.hexdigest()

def clean_id(series: pd.Series) -> pd.Series:
    return series.astype("string").str.strip()


def parse_boolish(value: object) -> bool | None:
    if pd.isna(value):
        return None
    text = str(value).strip().lower()
    if text in {"true", "1", "1.0"}:
        return True
    if text in {"false", "0", "0.0"}:
        return False
    raise ValueError(f"Unexpected CCBond value: {value!r}")


def aggregate_entanglement(series: pd.Series) -> float:
    """Repository rule: any covalent crossing excludes; otherwise NCLE if present."""
    values = [parse_boolish(value) for value in series]
    values = [value for value in values if value is not None]
    if any(values):
        return np.nan
    if values:
        return 1.0
    return 0.0


def read_fasta_ids(path: Path) -> tuple[list[str], set[str]]:
    ids: list[str] = []
    with path.open(errors="replace") as handle:
        for line in handle:
            if not line.startswith(">"):
                continue
            header = line[1:].strip()
            parts = header.split("|")
            accession = parts[1] if len(parts) >= 3 else header.split()[0]
            ids.append(accession.strip())
    return ids, set(ids)


def parse_regions(value: object) -> list[tuple[int, int]]:
    if pd.isna(value) or str(value).strip() in {"", "-"}:
        return []
    regions: list[tuple[int, int]] = []
    for token in str(value).split(";"):
        match = re.fullmatch(r"(\d+)-(\d+)", token.strip())
        if not match:
            raise ValueError(f"Unparseable disorder interval: {token!r}")
        start, end = map(int, match.groups())
        if start < 1 or end < start:
            raise ValueError(f"Invalid disorder interval: {token!r}")
        regions.append((start, end))
    return regions


def compare_series(left: pd.Series, right: pd.Series) -> pd.Series:
    if pd.api.types.is_numeric_dtype(left) and pd.api.types.is_numeric_dtype(right):
        return pd.Series(
            ~np.isclose(left, right, equal_nan=True, atol=1e-12, rtol=1e-12),
            index=left.index,
        )
    return left.fillna("<NA>").astype(str).ne(right.fillna("<NA>").astype(str))

def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as handle:
        for chunk in iter(lambda: handle.read(1024 * 1024), b""):
            digest.update(chunk)
    return digest.hexdigest()

    candidate_mappings["Header_candidates_pLDDT_ge70"] = candidate_mappings[
        "Header_Jiang_candidates"
    ].map(lambda values: sorted(set(values).intersection(plddt_ge70_ids)))
    candidate_mappings["n_candidates_all_pLDDT"] = candidate_mappings[
        "Header_Jiang_candidates"
    ].map(len)
    candidate_mappings["n_candidates_pLDDT_ge70"] = candidate_mappings[
        "Header_candidates_pLDDT_ge70"
    ].map(len)
    candidate_mappings["Header_Jiang_candidates"] = candidate_mappings[
        "Header_Jiang_candidates"
    ].map(";".join)
    candidate_mappings["Header_candidates_pLDDT_ge70"] = candidate_mappings[
        "Header_candidates_pLDDT_ge70"
    ].map(";".join)

    # Manual source-to-output traces: four final rows per quadrant plus exclusions.
    traced_ids: list[str] = []
    for group in group_order:
        traced_ids.extend(
            complete.loc[complete["Analysis_group"].eq(group), "UniProt_ID"].sort_values().head(4)
        )

    def first_id(frame: pd.DataFrame) -> str | None:
        values = frame["UniProt_ID"].sort_values()
        return values.iloc[0] if len(values) else None

    extras = [
        first_id(edned.loc[edned["Entanglement"].isna()]),
        first_id(ent_eligible.loc[ent_eligible["Knotted"]]),
        first_id(knot_free.loc[~knot_free["FASTA_available"]]),
        first_id(fasta_complete.loc[fasta_complete["Membrane"]]),
    ]
    for group in group_order:
        extras.append(
            first_id(core.loc[core["Analysis_group"].eq(group) & core["Fraction_IDP_IDR"].isna()])
        )
    extras.append("Q8NI36" if "Q8NI36" in set(master["UniProt_ID"]) else None)
    traced_ids.extend(value for value in extras if value is not None)
    traced_ids = list(dict.fromkeys(traced_ids))

    trace = master.loc[master["UniProt_ID"].isin(traced_ids)].copy()
    trace = trace.merge(ent_diag, on="UniProt_ID", how="left", suffixes=("", "_source"))
    trace["In_ED_NED"] = trace["Degradation_profile"].isin(["ED", "NED"])
    trace["Entanglement_eligible"] = trace["Entanglement"].notna()
    trace["Knot_free"] = ~trace["Knotted"]
    trace["Non_membrane"] = ~trace["Membrane"]
    trace["MobiDB_complete"] = trace["Fraction_IDP_IDR"].notna()
    trace["Analysis_group"] = np.where(
        trace["Degradation_profile"].isin(["ED", "NED"]) & trace["Entanglement"].notna(),
        trace["Degradation_profile"]
        + "_"
        + trace["Entanglement"].map({1.0: "Entangled", 0.0: "NotEntangled"}),
        pd.NA,
    )

    def exclusion_reason(row: pd.Series) -> str:
        if not row["In_ED_NED"]:
            return "not ED/NED"
        if not row["Entanglement_eligible"]:
            return "covalent-lasso entanglement"
        if not row["Knot_free"]:
            return "knotted"
        if not row["FASTA_available"]:
            return "FASTA accession unavailable"
        if not row["Non_membrane"]:
            return "membrane keyword"
        if not row["MobiDB_complete"]:
            return "missing MobiDB fraction"
        return "included in 606"

    trace["Final_result"] = trace.apply(exclusion_reason, axis=1)
    trace["Trace_order"] = trace["UniProt_ID"].map({value: i + 1 for i, value in enumerate(traced_ids)})
    trace = trace.sort_values("Trace_order")
    trace_columns = [
        "Trace_order",
        "UniProt_ID",
        "Average_pLDDT",
        "pLDDT_SD",
        "Degradation_profile",
        "Bi_exponential_decay",
        "ENT_ID_nonmissing",
        "CCBond_true",
        "CCBond_false",
        "CCBond_missing",
        "Entanglement",
        "Entanglement_status",
        "Knotted",
        "FASTA_available",
        "Membrane",
        "Disorder_query_status",
        "Disorder_residues",
        "MobiDB_length",
        "Fraction_IDP_IDR",
        "Analysis_group",
        "Final_result",
    ]
    trace = trace.reindex(columns=trace_columns)

    # Merge/source diagnostics.
    raw_ccbond_values = Counter(str(value).strip() for value in ent_raw["CCBond"].dropna())
    invalid_ccbond = sum(
        count
        for value, count in raw_ccbond_values.items()
        if value.lower() not in {"true", "false", "1", "0", "1.0", "0.0"}
    )
    exact_edned_matches = int(experimental_edned["Exact_Jiang_match"].sum())
    no_exact = len(experimental_edned) - exact_edned_matches
    has_header_candidate = int(
        experimental_edned.loc[~experimental_edned["Exact_Jiang_match"], "Header_Jiang_candidates"]
        .map(bool)
        .sum()
    )
    merge_audit = pd.DataFrame(
        [
            {
                "source": "McShane",
                "rows": len(mcshane),
                "unique_IDs": mcshane["UniProt_ID"].nunique(),
                "missing_IDs": mcshane["UniProt_ID"].isna().sum(),
                "duplicate_IDs": mcshane["UniProt_ID"].duplicated().sum(),
                "coverage_or_note": "All rows have one exact Protein IDs (Uniprot) value",
            },
            {
                "source": "Jiang pLDDT",
                "rows": len(plddt),
                "unique_IDs": plddt["UniProt_ID"].nunique(),
                "missing_IDs": plddt["UniProt_ID"].isna().sum(),
                "duplicate_IDs": plddt["UniProt_ID"].duplicated().sum(),
                "coverage_or_note": f"{len(plddt_master)} pass mean pLDDT >= 70",
            },
            {
                "source": "Jiang entanglement",
                "rows": len(ent_raw),
                "unique_IDs": ent_raw["UniProt_ID"].nunique(),
                "missing_IDs": ent_raw["UniProt_ID"].isna().sum(),
                "duplicate_IDs": len(ent_raw) - ent_raw["UniProt_ID"].nunique(),
                "coverage_or_note": f"{invalid_ccbond} invalid nonmissing CCBond values",
            },
            {
                "source": "processed MobiDB/master export",
                "rows": len(provided),
                "unique_IDs": provided["UniProt_ID"].nunique(),
                "missing_IDs": provided["UniProt_ID"].isna().sum(),
                "duplicate_IDs": provided["UniProt_ID"].duplicated().sum(),
                "coverage_or_note": f"{provided['Fraction_IDP_IDR'].notna().sum()} nonmissing fractions",
            },
            {
                "source": "human proteome FASTA",
                "rows": len(fasta_list),
                "unique_IDs": len(fasta_ids),
                "missing_IDs": 0,
                "duplicate_IDs": len(fasta_list) - len(fasta_ids),
                "coverage_or_note": f"{master['FASTA_available'].sum()} of master IDs matched",
            },
            {
                "source": "membrane-keyword FASTA",
                "rows": len(membrane_list),
                "unique_IDs": len(membrane_ids),
                "missing_IDs": 0,
                "duplicate_IDs": len(membrane_list) - len(membrane_ids),
                "coverage_or_note": f"{master['Membrane'].sum()} of master IDs matched",
            },
            {
                "source": "AlphaKnot human records",
                "rows": len(human_knots),
                "unique_IDs": human_knots["UniProt_ID"].nunique(),
                "missing_IDs": human_knots["UniProt_ID"].isna().sum(),
                "duplicate_IDs": human_knots["UniProt_ID"].duplicated().sum(),
                "coverage_or_note": f"{master['Knotted'].sum()} of master IDs matched after -F# removal",
            },
            {
                "source": "McShane ED/NED -> Jiang exact accession",
                "rows": len(experimental_edned),
                "unique_IDs": experimental_edned["UniProt_ID"].nunique(),
                "missing_IDs": 0,
                "duplicate_IDs": 0,
                "coverage_or_note": (
                    f"{exact_edned_matches} exact; {no_exact} unmatched; "
                    f"{has_header_candidate} unmatched records contain >=1 possible Jiang accession in grouped FASTA header"
                ),
            },
        ]
    )

    status = master["Disorder_query_status"].value_counts(dropna=False)
    observed_benchmarks = {
        "McShane rows": len(mcshane),
        "McShane ED": int(mcshane["Degradation_profile_raw"].eq("ED").sum()),
        "McShane NED": int(mcshane["Degradation_profile_raw"].eq("NED").sum()),
        "McShane UN": int(mcshane["Degradation_profile_raw"].eq("UN").sum()),
        "ED/NED rows": int(mcshane["Degradation_profile_raw"].isin(["ED", "NED"]).sum()),
        "pLDDT source proteins": len(plddt),
        "pLDDT >= 70 master": len(master),
        "Native entangled in master": int(master["Entanglement"].eq(1).sum()),
        "Not entangled in master": int(master["Entanglement"].eq(0).sum()),
        "Covalent-lasso excluded in master": int(master["Entanglement"].isna().sum()),
        "ED/NED in master": len(edned),
        "Eligible entanglement among ED/NED": len(ent_eligible),
        "After knot exclusion": len(knot_free),
        "FASTA-complete pre-membrane cohort": len(fasta_complete),
        "Non-membrane core": len(core),
        "MobiDB-complete core": len(complete),
        "MobiDB ok": int(status.get("ok", 0)),
        "MobiDB no selected annotation": int(status.get("mobidb_no_selected_annotation", 0)),
        "MobiDB no selected annotation and missing length": int(
            status.get("mobidb_no_selected_annotation|uniprot_missing_length", 0)
        ),
        "MobiDB missing length": int(status.get("uniprot_missing_length", 0)),
        "Knotted proteins in master (reported)": int(master["Knotted"].sum()),
        "FASTA available in master": int(master["FASTA_available"].sum()),
    }
    benchmark_comparison = pd.DataFrame(
        [
            {

                "difference": observed_benchmarks[name] - expected,
                "matches": observed_benchmarks[name] == expected,

    quality_checks = pd.DataFrame(
        [
            ("provided master duplicate IDs", int(master["UniProt_ID"].duplicated().sum()), 0),

        columns=["check", "observed", "expected_or_minimum"],

    # Workbook-level comparisons can only run when exact files are supplied.
    workbook_rows: list[dict[str, object]] = []
    expected_frames = {

        "mobidb_complete": complete,

    for label, filename in TARGET_WORKBOOKS.items():
        matches = find_any(workspace, filename)
        if not matches:
            workbook_rows.append(

                    "status": "MISSING",
                    "rows": np.nan,
                    "unique_IDs": np.nan,

        book = pd.read_excel(matches[0])
        id_column = next((column for column in id_candidates if column in book.columns), None)

                    "status": "ID COLUMN NOT FOUND",

        workbook_rows.append(
            {
                "workbook": filename,
                "status": "FOUND",
                "rows": len(book),
                "unique_IDs": len(book_ids),
                "expected_rows": len(expected_frames[label]),
                "ID_set_differences": len(book_ids.symmetric_difference(expected_ids)),
                "note": str(matches[0]),
            }

        "nonmembrane_core_rows": len(core),
        "mobidb_complete_rows": len(complete),
        "all_source_derived_master_fields_match_processed_export": bool(
            source_field_comparison["mismatches"].eq(0).all()
        ),
        "reported_knot_count": 123,
        "independent_knot_count": int(master["Knotted"].sum()),
        "mobidb_displayed_interval_discrepancies": len(interval_discrepancies),
        "mobidb_discrepancies_in_606": int(interval_discrepancies["in_606_complete"].sum()),
        "multi_track_mobidb_rows": int(
            master["MobiDB_annotation_tracks"].astype(str).str.contains(";", regex=False).sum()
        ),
        "target_workbooks_missing": int(workbook_comparison["status"].eq("MISSING").sum()),
        "manual_trace_rows": len(trace),
        "secretory_filter_applied": False,
        "association_testing_performed": False,
    }
    (output / "audit_summary.json").write_text(json.dumps(summary, indent=2) + "\n")
    print(json.dumps(summary, indent=2))
